    {% for r in rows %}
  • {{ r[label_key] }}{{ r[value_key] if r[value_key] is not none else '–' }}{{ suffix if r[value_key] is not none }}
    • {% endfor %}
{% endmacro %} {% macro total_line(s) %} {% set cum = s.cumulative %}{% set cmax = (cum|map(attribute='total')|max) or 1 %}{% set n = cum|length %} {% endmacro %}
